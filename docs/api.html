API Reference

Core Engine

from core.engine import CoreEngine

engine = CoreEngine()
engine.start()
engine.stop()

Scanner API

from run import GlanderMain

scanner = GlanderMain(target_url="https://target.com")
results = scanner.run()

Exploit Engine

from exploits.engine import ExploitEngine

exploit = ExploitEngine()
result = exploit.run_xss("https://target.com")
result = exploit.run_sqli("https://target.com")

Telemetry API

from telemetry.engine import TelemetryEngine

telemetry = TelemetryEngine()
telemetry.start()
telemetry.collect({"data": "value"})

Local Storage API

from local_storage.engine import LocalStorageEngine

storage = LocalStorageEngine()
storage.start()
storage.db.insert({"test": "data"})

Auth API

from auth.engine import AuthEngine

auth = AuthEngine()
auth.authenticate("user", "password")
auth.generate_token("user")

Proxy API

from proxy.engine import ProxyEngine

proxy = ProxyEngine()
proxy.add("192.168.1.1", 8080)
best = proxy.get_best()
