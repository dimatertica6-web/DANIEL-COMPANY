Examples

Example 1: Full Website Scan

from run import GlanderMain

scanner = GlanderMain(target_url="https://example.com")
results = scanner.run()
print(f"Found {results['total_pages']} pages")
print(f"Vulnerabilities: XSS={results['total_xss']}, SQLi={results['total_sqli']}")

Example 2: Run XSS Exploit

from exploits.xss_exploit import XSSExploit

xss = XSSExploit()
result = xss.execute("https://example.com/search?q=test")
if result["vulnerable"]:
    print("XSS vulnerability found!")
    print(f"Payload: {result['payloads'][0]}")

Example 3: Start Telemetry Server

from telemetry.engine import TelemetryEngine

server = TelemetryEngine()
server.start()
print("Telemetry server running on port 5555")
# Send data
server.collector.process({"user": "test", "action": "scan"})

Example 4: Local Storage

from local_storage.engine import LocalStorageEngine

storage = LocalStorageEngine()
storage.start()
storage.db.insert({"name": "John", "age": 30})
data = storage.db.get_all()
print(data)

Example 5: Authentication

from auth.engine import AuthEngine

auth = AuthEngine()
auth.initialize()
result = auth.create_user("testuser", "SecurePass123!")
if result["success"]:
    token = auth.authenticate("testuser", "SecurePass123!")
    print(f"Token: {token}")
